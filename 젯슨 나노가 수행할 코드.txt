import time
import requests
import cv2

SERVER_URL = "http://127.0.0.1:8000" # 실제 배포 시 FastAPI 서버 IP로 변경
DEVICE_ID = "jetson-nano-01"

def send_heartbeat():
    try:
        res = requests.post(f"{SERVER_URL}/api/device/heartbeat", 
                            json={"device_id": DEVICE_ID, "camera_ready": True}, timeout=2)
        return res.json().get("expected_phase")
    except Exception as e:
        print(f"서버 연결 실패: {e}")
        return None

def upload_image(image_path, phase):
    print(f"[{phase.upper()}] 이미지 서버 업로드 중...")
    try:
        with open(image_path, "rb") as f:
            # phase 폼 데이터와 파일을 함께 전송
            res = requests.post(f"{SERVER_URL}/api/capture", 
                                data={"phase": phase}, 
                                files={"file": f})
            print("업로드 성공:", res.json())
    except Exception as e:
        print(f"업로드 실패: {e}")

# =====================================
# 메인 루프 (카메라 촬영 시뮬레이션)
# =====================================
cap = cv2.VideoCapture(0) # 실제 카메라 인덱스 지정

while True:
    expected_phase = send_heartbeat()
    
    # 풋스위치 등을 눌렀을 때의 조건 (여기서는 임시로 엔터키로 가정)
    # 실제 환경에서는 GPIO 핀 입력으로 변경
    user_input = input(f"현재 대기상태: {expected_phase} | 촬영하려면 엔터 (종료: q): ")
    
    if user_input.lower() == 'q':
        break
        
    if expected_phase in ["before", "after"]:
        ret, frame = cap.read()
        if ret:
            img_name = f"capture_{expected_phase}.jpg"
            cv2.imwrite(img_name, frame)
            upload_image(img_name, expected_phase)
            
            # 수술 전(before)을 찍었다면, 30초 뒤 자동으로 수술 후(after) 촬영 진행
            if expected_phase == "before":
                print("수술 진행 중... (30초 후 After 자동 촬영)")
                time.sleep(30)
                
                ret, frame_after = cap.read()
                if ret:
                    cv2.imwrite("capture_after.jpg", frame_after)
                    upload_image("capture_after.jpg", "after")
        else:
            print("카메라에서 영상을 읽어올 수 없습니다.")

cap.release()
